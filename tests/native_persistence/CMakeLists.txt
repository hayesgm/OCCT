cmake_minimum_required(VERSION 3.20)
project(OCCTNativePersistenceTests LANGUAGES CXX)
enable_testing()
set(OCCT_ROOT "" CACHE PATH "OCCT installation to test")
if(NOT IS_ABSOLUTE "${OCCT_ROOT}")
  message(FATAL_ERROR "Set OCCT_ROOT to the absolute installation prefix")
endif()
find_path(OCCT_TEST_INCLUDE gp_Trsf.hxx PATHS "${OCCT_ROOT}/include/opencascade" NO_DEFAULT_PATH REQUIRED)
foreach(toolkit TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO)
  find_library(OCCT_TEST_${toolkit} NAMES ${toolkit} PATHS "${OCCT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  list(APPEND OCCT_TEST_LIBRARIES "${OCCT_TEST_${toolkit}}")
endforeach()
foreach(test stored_transforms analytic_readers projection_controls)
  add_executable(${test} ${test}.cxx)
  target_compile_features(${test} PRIVATE cxx_std_17)
  target_include_directories(${test} PRIVATE "${OCCT_TEST_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../src/IntTools")
  target_link_libraries(${test} PRIVATE ${OCCT_TEST_LIBRARIES})
  set_target_properties(${test} PROPERTIES BUILD_RPATH "${OCCT_ROOT}/lib")
  add_test(NAME ${test} COMMAND ${test})
endforeach()
